import json
import random
import csv
from scrapy import Spider, Request
from scrapy.settings import Settings
from scrapy.crawler import CrawlerProcess
import urllib3
import re

# 禁用SSL警告
urllib3.disable_warnings(urllib3.exceptions.InsecureRequestWarning)

# 自定义设置
CUSTOM_SETTINGS = {
    'USER_AGENT': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
    'DOWNLOAD_DELAY': random.uniform(0.5, 1.5),
    'CONCURRENT_REQUESTS': 2,
    'CONCURRENT_REQUESTS_PER_DOMAIN': 2,
    'DOWNLOAD_TIMEOUT': 15,
    'LOG_LEVEL': 'INFO',
    'TELNETCONSOLE_ENABLED': False,
    'ROBOTSTXT_OBEY': False,
    'SSL_VERIFY': False,
    # 防止爬取过快被限制
    'AUTOTHROTTLE_ENABLED': True,
    'AUTOTHROTTLE_START_DELAY': 1,
    'AUTOTHROTTLE_MAX_DELAY': 5,
}

class DiscussionSpider(Spider):
    name = 'discussion'
    allowed_domains = ['discussion.fedoraproject.org']
    start_urls = []
    all_data = []  # 存储所有爬取的数据
    target_count = 50  # 目标数据量
    current_page = 1  # 初始页码

    def start_requests(self):
        """生成初始列表页请求（从第1页开始）"""
        self.url_template = "https://discussion.fedoraproject.org/top?page={}&per_page=50&period=all"
        first_page_url = self.url_template.format(self.current_page)
        self.logger.info(f'开始爬取第 {self.current_page} 页：{first_page_url}')
        yield Request(
            url=first_page_url,
            callback=self.parse_topic_list,
            errback=self.handle_error,
            dont_filter=True
        )

    def parse_topic_list(self, response):
        """解析列表页HTML，严格按最新规则提取字段，达到50条则停止"""
        self.logger.info(f'正在解析第 {self.current_page} 页：{response.url} | 状态码：{response.status}')
        
        # 检查当前数据量是否已达标
        if len(self.all_data) >= self.target_count:
            self.logger.info(f'已达到目标数据量 {self.target_count} 条，停止解析当前页')
            return

        try:
            # 定位所有话题行
            topic_rows = response.xpath('//table[contains(@class, "topic-list")]/tbody/tr')
            self.logger.info(f'第 {self.current_page} 页发现 {len(topic_rows)} 个话题')

            for row_idx, row in enumerate(topic_rows):
                # 每次提取前先检查数据量，达标则立即停止
                if len(self.all_data) >= self.target_count:
                    self.logger.info(f'已达到目标数据量 {self.target_count} 条，停止提取话题')
                    break

                # ========== 严格按最新规则提取各字段 ==========
                # 1. 标题：从/t/xxx/xxx中提取xxx部分
                title_href = row.xpath('.//td[1]/span/a/@href').extract_first()
                title_value = None
                if title_href and '/t/' in title_href:
                    # 提取/t/和下一个/之间的内容（示例：latest-fedora-36-wallpaper-request-for-feedback）
                    title_match = re.search(r'/t/([^/]+)', title_href)
                    if title_match:
                        title_value = title_match.group(1)

                # 2. 主题类型：精准匹配class="badge-category__name"的span文本
                topic_type = row.xpath('.//td[1]/div/a/span/span[@class="badge-category__name"]/text()').extract_first()

                # 3. 标签：提取discourse-tags下的所有标签文本并拼接（示例：mindshare, design-team）
                tags = row.xpath('.//td[1]/div/div[@class="discourse-tags"]/a[@class="discourse-tag simple"]/text()').extract()
                tag_value = ', '.join(tags) if tags else None

                # 4. 作者：从img的title中提取-前的名称（示例：Máirín Duffy）
                author_title = row.xpath('.//td[2]/a[1]/img/@title').extract_first()
                author_value = None
                if author_title:
                    # 按第一个-分割，只取前面的作者名
                    author_parts = author_title.split(' - ', 1)
                    author_value = author_parts[0].strip() if author_parts else author_title.strip()

                # 5. 发布时间：提取td[5]的title属性中的创建日期精确值（示例：2022 年 3月 25 日 02:58）
                publish_time_title = row.xpath('.//td[5]/@title').extract_first()
                publish_time = None
                if publish_time_title:
                    # 提取"创建日期："后的精确时间（到换行符前）
                    time_match = re.search(r'创建日期：([^\n]+)', publish_time_title)
                    if time_match:
                        publish_time = time_match.group(1).strip()

                # 6. 总评论数：提取td[3]/a下的span.number文本（示例：19）
                comment_count = row.xpath('.//td[3]/a/span[@class="number"]/text()').extract_first()

                # 7. 阅读量：提取td[4]中span.number的title属性里的精确数字（示例：2952）
                view_title = row.xpath('.//td[4]/span[@class="number"]/@title').extract_first()
                view_count = None
                if view_title:
                    # 提取标题中的数字（去掉逗号，仅为获取精确值，非清洗）
                    view_match = re.search(r'(\d+(?:,\d+)*) 次', view_title)
                    if view_match:
                        view_count = view_match.group(1).replace(',', '')

                # 调试日志
                self.logger.debug(
                    f'第{self.current_page}页第{row_idx+1}行提取结果：\n'
                    f'标题：{title_value} | 主题类型：{topic_type} | 标签：{tag_value} | 作者：{author_value}\n'
                    f'发布时间：{publish_time} | 评论数：{comment_count} | 阅读量：{view_count}'
                )

                # 组装数据（保留所有原始提取值，不做任何清洗）
                topic_info = {
                    "标题": title_value,
                    "主题类型": topic_type,
                    "标签": tag_value,
                    "作者": author_value,
                    "发布时间": publish_time,
                    "总评论数": comment_count,
                    "阅读量": view_count
                }

                # 添加到数据列表
                self.all_data.append(topic_info)
                self.logger.info(f'提取话题成功 | 标题：{title_value or "无"} | 累计：{len(self.all_data)}/{self.target_count}')

            # 数据量未达标，请求下一页
            if len(self.all_data) < self.target_count:
                self.current_page += 1
                next_page_url = self.url_template.format(self.current_page)
                self.logger.info(f'第 {self.current_page-1} 页解析完成，累计 {len(self.all_data)} 条 | 请求下一页：{next_page_url}')
                yield Request(
                    url=next_page_url,
                    callback=self.parse_topic_list,
                    errback=self.handle_error,
                    dont_filter=True
                )
            else:
                self.logger.info(f'已达到目标数据量 {self.target_count} 条，停止请求下一页')

        except Exception as e:
            self.logger.error(f'列表页解析异常：{e} | 页码：{self.current_page}', exc_info=True)

    def handle_error(self, failure):
        """请求错误处理：记录错误，继续爬取下一页（如果数据量未达标）"""
        request = failure.request
        self.logger.error(f'请求失败：{request.url} | 错误类型：{failure.type} | 详情：{failure.value}', exc_info=True)
        
        # 失败后仍尝试请求下一页（避免单页失败导致整体停止）
        if len(self.all_data) < self.target_count:
            self.current_page += 1
            next_page_url = self.url_template.format(self.current_page)
            self.logger.info(f'第 {self.current_page-1} 页请求失败，尝试下一页：{next_page_url}')
            yield Request(
                url=next_page_url,
                callback=self.parse_topic_list,
                errback=self.handle_error,
                dont_filter=True
            )

    def closed(self, reason):
        """爬虫结束时保存 CSV（保留所有原始提取值）"""
        # 截断数据到目标数量（防止超额）
        final_data = self.all_data[:self.target_count]
        
        # 添加数据包编号
        for idx, item in enumerate(final_data, start=1):
            item["数据包编号"] = idx
        
        # 定义 CSV 表头（严格按指定字段顺序）
        csv_headers = [
            "数据包编号", "标题", "主题类型", "标签", 
            "作者", "发布时间", "总评论数", "阅读量"
        ]
        
        # 保存到 CSV
        try:
            # 确保目录存在
            import os
            save_path = 'discussion/fedora_centos_topics.csv'
            save_dir = os.path.dirname(save_path)
            if save_dir and not os.path.exists(save_dir):
                os.makedirs(save_dir)
            
            with open(save_path, mode='w', encoding='utf-8-sig', newline='') as f:
                writer = csv.DictWriter(f, fieldnames=csv_headers)
                writer.writeheader()
                writer.writerows(final_data)
            
            self.logger.info(f'\n爬虫结束原因：{reason}')
            self.logger.info(f'最终保存成功：共 {len(final_data)} 条数据（目标：{self.target_count} 条）')
            self.logger.info(f'文件路径：./{save_path}')
        except Exception as e:
            self.logger.error(f'保存 CSV 文件失败：{e}', exc_info=True)

def run_spider():
    settings = Settings()
    settings.setdict(CUSTOM_SETTINGS)
    # 可选：开启DEBUG日志（方便排查提取问题）
    # settings.set('LOG_LEVEL', 'DEBUG')
    process = CrawlerProcess(settings=settings)
    process.crawl(DiscussionSpider)
    process.start()
    print('=' * 50)
    print(f'爬取任务完成！最终获取 {len(DiscussionSpider.all_data)} 条数据')

if __name__ == '__main__':
    run_spider()
